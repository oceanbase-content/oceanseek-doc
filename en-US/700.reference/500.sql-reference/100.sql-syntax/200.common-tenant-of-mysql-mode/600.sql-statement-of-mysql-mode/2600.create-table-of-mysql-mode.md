# CREATE TABLE

## Purpose

This statement is used to create a new table in the database.

## Syntax

```sql
CREATE [hint_options] [TEMPORARY] TABLE [IF NOT EXISTS] table_name
      (table_definition_list) [table_option_list] [partition_option] [[MERGE_ENGINE = {delete_insert | partial_update}] table_column_group_option] [IGNORE | REPLACE] [AS] select;

CREATE TABLE [TEMPORARY] [IF NOT EXISTS] table_name
      LIKE table_name;

table_definition_list:
    table_definition [, table_definition ...]

table_definition:
      column_definition_list
    | [CONSTRAINT [constraint_name]] PRIMARY KEY index_desc
    | [CONSTRAINT [constraint_name]] UNIQUE {INDEX | KEY}
            [index_name] index_desc
    | [CONSTRAINT [constraint_name]] FOREIGN KEY
            [index_name] index_desc
            REFERENCES reference_definition
            [match_action][opt_reference_option_list]
    | [ASYNC] [FULLTEXT] {INDEX | KEY} [index_name] [index_type] (key_part,...) [WITH PARSER tokenizer_option] [PARSER_PROPERTIES[=](parser_properties_list)]
      [index_option_list] [index_column_group_option]
    | [ASYNC] SEARCH {INDEX | KEY} [index_name] (search_index_column [, search_index_column ...])
    | index_json_clause
    | [CONSTRAINT [constraint_name]] CHECK(expression) constranit_state

column_definition_list:
    column_definition [, column_definition ...]

column_definition:
     column_name data_type
         [DEFAULT const_value] [AUTO_INCREMENT]
         [NULL | NOT NULL] [[PRIMARY] KEY] [UNIQUE [KEY]] [COMMENT string_value] [SKIP_INDEX(skip_index_option_list)]
   | column_name data_type
         [GENERATED ALWAYS] AS (expr) [VIRTUAL | STORED]
         [opt_generated_column_attribute]

skip_index_option_list:
    skip_index_option [,skip_index_option ...]

skip_index_option:
    MIN_MAX
    | SUM

index_desc:
   (column_desc_list) [index_type] [index_option_list]

match_action:
   MATCH {SIMPLE | FULL | PARTIAL}

opt_reference_option_list:
   reference_option [,reference_option ...]

reference_option:
   ON {DELETE | UPDATE} {RESTRICT | CASCADE | SET NULL | NO ACTION | SET DEFAULT}

tokenizer_option:
    SPACE
    | NGRAM
    | BENG
    | IK
    | NGRAM2
    | analyzer

parser_properties_list:
    parser_properties, [parser_properties]

parser_properties:
    min_token_size = int_value
    | max_token_size = int_value
    | ngram_token_size = int_value
    | ik_mode = 'char_value'
    | min_ngram_size = int_value
    | max_ngram_size = int_value
    | analysis = json_string

key_part:
    {index_col_name [(length)] | (expr)} [ASC | DESC]

index_type:
    USING BTREE

index_option_list:
    index_option [ index_option ...]

index_option:
      [GLOBAL | LOCAL]
    | block_size
    | compression
    | STORING(column_name_list)
    | COMMENT string_value
    | STORAGE_CACHE_POLICY(storage_cache_policy_option)

table_option_list:
    table_option [ table_option ...]

table_option:
      [DEFAULT] {CHARSET | CHARACTER SET} [=] charset_name
    | [DEFAULT] COLLATE [=] collation_name
    | table_tablegroup
    | block_size
    | lob_inrow_threshold [=] num
    | compression
    | AUTO_INCREMENT [=] INT_VALUE
    | COMMENT string_value
    | ROW_FORMAT [=] REDUNDANT|COMPACT|DYNAMIC|COMPRESSED|DEFAULT
    | PCTFREE [=] num
    | parallel_clause
    | DUPLICATE_SCOPE [=] 'none|cluster'
    | TABLE_MODE [=] 'table_mode_value'
    | auto_increment_cache_size [=] INT_VALUE
    | READ {ONLY | WRITE}
    | ORGANIZATION [=] {INDEX | HEAP}
    | enable_macro_block_bloom_filter [=] {True | False}
    | DYNAMIC_PARTITION_POLICY [=] (dynamic_partition_policy_list)
    | SEMISTRUCT_ENCODING_TYPE [=] 'encoding' # Deprecated. Use SEMISTRUCT_PROPERTIES instead.
    | MICRO_BLOCK_FORMAT_VERSION [=] {1|2}
    | STORAGE_CACHE_POLICY (storage_cache_policy_option)
    | ASYNC_INDEX_PARAMS [=] 'async_index_params_string'


parallel_clause:
    {NOPARALLEL | PARALLEL integer}

table_mode_value:
    NORMAL
    | QUEUING
    | MODERATE
    | SUPER
    | EXTREME

dynamic_partition_policy_list:
    dynamic_partition_policy_option [, dynamic_partition_policy_option ...]

dynamic_partition_policy_option:
    ENABLE = {true | false}
    | TIME_UNIT = {'hour' | 'day' | 'week' | 'month' | 'year'}
    | PRECREATE_TIME = {'-1' | '0' | 'n {hour | day | week | month | year}'}
    | EXPIRE_TIME = {'-1' | '0' | 'n {hour | day | week | month | year}'}
    | TIME_ZONE = {'default' | 'time_zone'}
    | BIGINT_PRECISION = {'none' | 'us' | 'ms' | 's'}

partition_option:
      PARTITION BY HASH(expression)

    | PARTITION BY KEY([column_name_list])
      [subpartition_option] PARTITIONS partition_count
    | PARTITION BY RANGE {(expression) | COLUMNS (column_name_list)}
      [subpartition_option] (range_partition_list) [STORAGE_CACHE_POLICY = {"hot" | "auto" | "none"}]

      SUBPARTITIONS subpartition_count

      (range_subpartition_list) [STORAGE_CACHE_POLICY = {"hot" | "auto" | "none"}]
    | SUBPARTITION BY LIST(expression)


storage_cache_policy_option:
    GLOBAL = {"hot" | "auto" | "none"}
    | timeline_strategy_list

timeline_strategy_list:
    BOUNDARY_COLUMN = column_name [,BOUNDARY_COLUMN_UNIT = {"s"| "ms"}] ,HOT_RETENTON = intnum retention_time_unit

    YEAR
    | MONTH
    | WEEK
    | DAY
    | HOUR
    | MINUTE

    range_partition [, range_partition ...]

range_partition:
    PARTITION partition_name
    VALUES LESS THAN {(expression_list) | MAXVALUE}
    [STORAGE_CACHE_POLICY = {"hot" | "auto" | "none"}]

range_subpartition_list:
    range_subpartition [, range_subpartition ...]

range_subpartition:
    SUBPARTITION subpartition_name
    VALUES LESS THAN {(expression_list) | MAXVALUE}
    [STORAGE_CACHE_POLICY = {"hot" | "auto" | "none"}]

expression_list:
    expression [, expression ...]

column_name_list:
    column_name [, column_name ...]

partition_name_list:
    partition_name [, partition_name ...]

partition_count | subpartition_count:
    INT_VALUE

table_column_group_option/index_column_group_option:
      WITH COLUMN GROUP(all columns)
    | WITH COLUMN GROUP(each column)
    | WITH COLUMN GROUP(all columns, each column)

index_json_clause:
    [UNIQUE] INDEX idx_json_name((CAST(json_column_name->'$.json_field_name' AS UNSIGNED ARRAY)))
    | INDEX idx_json_name(column_name, [column_name, ...] (CAST(json_column_name->'$.json_field_name' AS CHAR(n) ARRAY)))
```

### table_mode_value

<main id="notice" type='explain'>
  <h4>Note</h4>
  <p>In the <code>TABLE_MODE</code> modes listed below, all modes except for <code>NORMAL</code> mode represent <code>QUEUING</code> tables. This type of <code>QUEUING</code> table is the most basic table type, and the several modes listed subsequently (except for NORMAL mode) represent more aggressive major compaction strategies.</p>
</main>

* `NORMAL`: The default value, indicating normal. In this mode, the probability of triggering a major compaction after a minor compaction is extremely low.
* `QUEUING`: In this mode, the probability of triggering a major compaction after a minor compaction is low.
* `MODERATE`: Indicates moderate. In this mode, the probability of triggering a major compaction after a minor compaction is medium.
* `SUPER`: Super. In this mode, a major compaction is triggered after a minor compaction with a high probability.
* `EXTREME`: Extreme. In this mode, a major compaction is triggered after a minor compaction with a higher probability.

For more information about major compactions, see [Adaptive compaction](../../../../200.system-management/500.manage-data-storage/200.merge-management/320.adaptive-compaction.md).

  | min_token_size   | [1, 16] |
  | max_token_size   | [10, 84]|

* `NGRAM`: Indicates the word segmentation method based on N-gram (Chinese). The following attributes can be specified:

  |        Attribute      | Value range |
  |------------------|---------|
  | ngram_token_size | [1, 10] |

* `NGRAM2`: Indicates that the text should be split into consecutive characters within the range of `min_ngram_size` to `max_ngram_size`. The following attributes can be specified:

  |        Attribute      | Value range |
  |------------------|---------|
  | min_ngram_size   | [1, 16] |
  | max_ngram_size   | [1, 16] |

* `BENG`: word segmentation method based on Beng (Basic English). The following attributes can be specified:

  |        Attribute      | Value range |
  |------------------|---------|
  | min_token_size   | [1, 16] |
  | max_token_size   | [10, 84]|

* `IK`: Indicates the word segmentation method based on IK (Chinese). Currently, only the `utf-8` character set is supported. The following attributes can be specified:

  |        Attribute      | Value range |
  |------------------|---------|
  | ik_mode          | <ul><li>`smart`</li><li>`max_word`</li></ul>|

* `analyzer`: the text analyzer. It is used through the `analysis` parameter in `PARSER_PROPERTIES`.

### parser_properties

* `min_token_size`: The minimum token length. The default value is 3, and the valid range is 1 to 16.

* `max_token_size`: The maximum token length. Default value: 84. Value range: 10 to 84.

* `ngram_token_size`: The token length for `NGRAM`. This parameter is valid only when the tokenizer is `NGRAM`. The default value is 2, and the valid range is 1 to 10.

* `ik_mode`: The stemming mode of the `IK` stemmer. Valid values:

  * `smart`: The default value. This mode prioritizes identifying longer words, reducing the number of splits and ensuring each character belongs to only one word. It also attempts to merge numerals with measure words into a single word in the output.
  * `max_word`: Outputs recognizable words as much as possible. Longer words and the shorter words within them are retained. The same character can appear in multiple words.

* `min_ngram_size`: Only valid for the `NGRAM2` tokenizer. Specifies the minimum number of characters for a generated n-gram. Value range: `[1, 16]`.

* `max_ngram_size`: Valid only for the `NGRAM2` tokenizer. Specifies the maximum number of characters for a generated n-gram. The value range is `[1, 16]`. The tokenizer generates consecutive character segments with lengths between the minimum and maximum values.

* `analysis`: Valid only when `WITH PARSER analyzer` is specified. Specifies the JSON definition string for the analyzer. Built-in analyzers can be abbreviated to forms such as `{"analyzer": "standard"}`. For a complete list of built-in types and their corresponding languages, see the description of the `analyzer_type` parameter in [TOKENIZE](../400.functions-of-mysql-mode/960.text-processing-functions-of-mysql-mode/200.tokenize-of-mysql-mode.md).

**Example:**

```sql
CREATE TABLE tbl1 (col1 VARCHAR(200), col2 TEXT,
    FULLTEXT INDEX fidx(col2) WITH PARSER ik PARSER_PROPERTIES=(ik_mode='max_word'));
```

```sql
CREATE TABLE docs (
    id BIGINT PRIMARY KEY,
    content TEXT,
    FULLTEXT INDEX ft_idx_content(content)
        WITH PARSER analyzer
        PARSER_PROPERTIES=(analysis='{"analyzer": "standard"}')
);
```

### dynamic_partition_policy_option

* `ENABLE = {true | false}`: Optional. Specifies whether to enable dynamic partition management. Modifiable. Valid values:

  * `true`: The default value. It specifies to enable dynamic partition management.
  * `false`: Indicates that dynamic partition management is disabled.

* `TIME_UNIT = {'hour' | 'day' | 'week' | 'month' | 'year'}`: The time unit of the partition, which is the interval for automatically creating partition boundaries. This parameter is required and cannot be modified. Valid values:

  * `hour`: Partitions are divided by hour.
  * `day`: Partitions are divided by day.
  * `week`: Partitions are divided by week.
  * `month`: Partitions are divided by month.
  * `year`: Partitions are divided by year.

* `PRECREATE_TIME = {'-1' | '0' | 'n {hour | day | week | month | year}'}`: Optional. Specifies the time for precreating partitions. When you call this API to schedule dynamic partition management, partitions are precreated so that **max_partition_upper_bound > now() + precreate_time**. The value can be modified. Valid values:

  * `-1`: The default value, which indicates not to pre-create partitions.
  * `0`: Indicates that only the current partition is to be pre-created.
  * `n {hour | day | week | month | year}`: Indicates to pre-create partitions for the specified time span. For example, `3 hour` means to pre-create partitions for the next 3 hours.

  <main id="notice" type='explain'>
    <h4>Note</h4>
    <p><ul><li>If you need to pre-create multiple partitions, the interval between partition boundaries is <code>TIME_UNIT</code>.</li><li>The boundary of the first pre-created partition is the existing largest partition boundary rounded up by <code>TIME_UNIT</code>.</li></ul></p>
  </main>

* `EXPIRE_TIME = {'-1' | '0' | 'n {hour | day | week | month | year}'}`: Optional. Specifies the partition expiration time. When you schedule a dynamic partition management task, all partitions that meet the condition **partition upper bound < now() - expire_time** will be deleted. The value can be modified. Valid values:

  * `-1`: the default value, which indicates that the partition never expires.
  * `0`: Indicates that all partitions except the current one have expired.
  * `n {hour | day | week | month | year}`: The partition expiration time. For example, `1 day` indicates the partition expires in 1 day.

* `TIME_ZONE = {'default' | 'time_zone'}`: Optional. Specifies the time zone information used when comparing the current time with the partitioning key of a time type (`date`, `datetime`, or `year`) value. This parameter cannot be modified. Valid values:

  * `default`: the default value, which indicates that no additional time zone is configured and the tenant's time zone is used. For all time zones other than those listed above, the `time_zone` field must be set to `default`.
  * `time_zone`: The custom time zone offset, for example, `+8:00`.

* `BIGINT_PRECISION = {'none'| 'ms' | 's'}`: Optional. Specifies the timestamp precision of the partitioning key of the `bigint` type. This parameter cannot be modified. Valid values:

  * `none`: The default value, indicating no precision (the partitioning key is not of the `bigint` type).
  * `ms`: millisecond precision.
  * `s`: seconds.

For more information about creating a dynamic partitioned table, see [Create a dynamic partitioned table](../../../../300.database-object-management/300.manage-partitions-of-mysql-mode/260.dynamic-partition-of-mysql-mode/200.create-a-dynamic-partition-table-of-mysql-mode.md).

**Example:**

```sql
CREATE TABLE tbl2 (col1 INT, col2 DATETIME)
    DYNAMIC_PARTITION_POLICY(
        ENABLE = true,
        TIME_UNIT = 'hour',
        PRECREATE_TIME = '3 hour',
        EXPIRE_TIME = '1 day',
        TIME_ZONE = '+8:00',
        BIGINT_PRECISION = 'none')
    PARTITION BY RANGE COLUMNS (col2)(
        PARTITION P0 VALUES LESS THAN ('2024-11-11 13:30:00')
        );
```

### storage_cache_policy

#### Create table-level storage_cache_policy

The `STORAGE_CACHE_POLICY` option is optional. If you do not specify the `STORAGE_CACHE_POLICY` option, the value of the tenant parameter [default_storage_cache_policy](../../../../800.configuration-items-and-system-variables/100.system-configuration-items/400.tenant-level-configuration-items/2120.default_storage_cache_policy.md) is used by default during table creation. The `storage_cache_policy_option` defines different attributes in key-value pairs. The semantics of each attribute are as follows:

* `GLOBAL = {"hot"|"auto"|"none"}`: Specifies the hotspot caching strategy for all table data. Valid values:

  * `hot`: Indicates that all data in the specified table is hot data. If sufficient cache disk space is available, all data under the table will be cached to the local cache disk.
  * `auto`: Indicates that the hot data for the specified table is automatically identified by the system.
  * `none`: Indicates that the index follows the value of the primary table's `STORAGE_CACHE_POLICY`.

    <main id="notice" type='notice'>
      <h4>Notice</h4>
      <p>The <code>none</code> attribute can only be used for indexes.</p>
    </main>
* `timeline_strategy_list`: The list of timeline strategy parameters, separated by commas. The hotspot cache timeline strategy uses time to determine whether partition data is hot data. Based on the configured strategy, the system automatically adjusts the partition data on the local cache disk.

  <main id="notice" type='explain'>
    <h4>Note</h4>
    <p>Timeline: Hot data is cached based on the partitioning range defined by RANGE partitioning. When certain time conditions are met, the data in that partition is identified as hot data. When using the timeline strategy, note the following considerations:<ul><li>Only supported for RANGE-partitioned tables (RANGE partitioning can be at the primary or secondary level), because time is required to determine whether a partition has expired. However, it cannot be a dual RANGE partition, as it would be impossible to determine which RANGE partition's time to use.</li><li>When using the timeline strategy, the partition expression can only contain column names; expression conversion is not supported. For example, <code>PARTITION BY RANGE COLUMNS(expr(col3))</code>.</li><li><code>BOUNDARY_COLUMN</code> must be the partitioning key. If the partitioning key consists of multiple columns, <code>BOUNDARY_COLUMN</code> must be the first column, used to determine whether the partition has expired.</li></ul></p>
  </main>

* `BOUNDARY_COLUMN = column_name`: Identifies the column for hot data detection. Supports integer types (`BIGINT` or `INT`, in Unix timestamp format) and time types (`TIMESTAMP`, `DATE`, `DATETIME`, or `YEAR`).

  * If `BOUNDARY_COLUMN` is of the integer type, the table supports RANGE and RANGE COLUMNS partitioning.
  * If the `BOUNDARY_COLUMN` type is a time type, the table can only use the RANGE COLUMNS partitioning method.

  * `BOUNDARY_COLUMN_UNIT = {"s" | "ms"}`: specifies the time unit for the `BOUNDARY_COLUMN` parameter. This parameter is supported only when `BOUNDARY_COLUMN` is an integer. When an integer value is used as a timestamp, a timestamp unit must be specified; otherwise, it may lead to incorrect time parsing. Valid values are as follows:

    * When the partitioning column is of type `INT`, the value of `BOUNDARY_COLUMN_UNIT` can only be `s`.
    * When the partitioning column is of the `BIGINT` type, the value of `BOUNDARY_COLUMN_UNIT` can be `s` or `ms`.

      <main id="notice" type='notice'>
        <h4>Notice</h4>
        <p>If the format is not a Unix timestamp, the time represented by the <code>INT</code> type cannot be correctly identified.</p>
      </main>

* `HOT_RETENTION = intnum retention_time_unit`: Specifies the time range for hot data.

  * `intnum`: an integer.
  * `retention_time_unit`: The time unit. Valid values:

    * `YEAR`: year.
    * `MONTH`: the month.
    * `WEEK`: week.
    * `DAY`: the number of days.
    * `HOUR`: the hour.
    * `MINUTE`: the number of minutes.

#### Create partition-level storage_cache_policy

The partition-level `STORAGE_CACHE_POLICY` is optional and can have the following values:

* `hot`: Indicates that all data in the specified partition is hot data. If sufficient cache disk space is available, all data in the partition will be cached to the local cache disk.
* `auto`: Indicates that the system automatically identifies the hot data for the specified partition.
* `none`: The default value. It indicates that the hotspot cache policy for this partition follows the value of `STORAGE_CACHE_POLICY` of its parent partition.

  ```sql
  ) PARTITION BY RANGE (c1)
  (PARTITION p1 VALUES LESS THAN (100) STORAGE_CACHE_POLICY = "hot",
  PARTITION p2 VALUES LESS THAN (200) STORAGE_CACHE_POLICY = "auto",
  PARTITION p3 VALUES LESS THAN MAXVALUE STORAGE_CACHE_POLICY = "none");
  ```

  Example description:
  * When creating table `tbl5`, two indexes were created simultaneously, each specifying a different storage caching strategy.
  * The table is partitioned by the `c1` column using RANGE partitioning, and each partition has a specified storage and caching strategy.
  * The STORAGE_CACHE_POLICY for indexes and table partitions can be set independently, enabling fine-grained storage management.

### Create an asynchronous index

Example of creating an asynchronous full-text index:

```sql
CREATE TABLE doc_async (
    c1 INT,
    content VARCHAR(255),
    ASYNC FULLTEXT INDEX idx_content(content)
) ORGANIZATION HEAP;
```

Example of creating an async search index:

```sql
CREATE TABLE t (
    id BIGINT,
    doc JSON,
    ASYNC SEARCH INDEX idx_search (doc)
) ORGANIZATION HEAP ASYNC_INDEX_PARAMS = 'ASYNC_REFRESH_INTERVAL=10';
```
